### `scripts/secrets-grep.mjs`

```js
#!/usr/bin/env node
/** Records security.test.mjs discipline, ported (NAV-007, Security and secret hygiene).
 *  Fails on live-secret patterns anywhere in the repository. */
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";

const PATTERNS = [
  [/sk_live_[A-Za-z0-9]{20,}/, "Stripe live key"],
  [/re_[A-Za-z0-9]{30,}/, "Resend API key"],
  [/-----BEGIN (RSA |EC )?PRIVATE KEY-----/, "private key material"],
  [/pepper\s*[:=]\s*["'][^"']{8,}/i, "session pepper"],
  [/[A-Za-z0-9_-]{40}\.[A-Za-z0-9_-]{14,}\.[A-Za-z0-9_-]{20,}/, "JWT-shaped secret"],
];
const SKIP = new Set([".git", "node_modules", ".next", ".turbo", ".wrangler"]);

let failures = 0;
function walk(dir) {
  for (const entry of readdirSync(dir)) {
    if (SKIP.has(entry)) continue;
    const p = join(dir, entry);
    const s = statSync(p);
    if (s.isDirectory()) walk(p);
    else if (/\.(ts|tsx|js|mjs|cjs|json|jsonc|toml|ya?ml|md|html|css|env.*)$/.test(entry) || entry.startsWith(".env")) {
      const text = readFileSync(p, "utf8");
      for (const [re, label] of PATTERNS)
        if (re.test(text)) { console.error(`✗ ${label}: ${p}`); failures++; }
    }
  }
}
walk(".");
if (failures) { console.error(`${failures} secret-pattern hit(s)`); process.exit(1); }
console.log("secrets-grep: clean");
```

Bring-up note: the draft's `(?i)pepper…` pattern was a syntax error at load time — JavaScript has no inline flags; the `/i` flag replaces it.

### `scripts/verify-bindings.mjs`

```js
#!/usr/bin/env node
/** NAV-006 (Service bindings are non-inheritable): parse every apps/<app>/wrangler.jsonc; assert service
 *  bindings follow <app>-worker-<env> and target a known Worker.
 *  (bring-up: the draft comment wrote apps/<star>/wrangler.jsonc, whose
 *  <star>/ closed the block comment early — reworded, not re-scoped.) */
import { readdirSync, readFileSync, existsSync } from "node:fs";
import { join } from "node:path";

const APPS = "apps";
let failures = 0;

// bring-up: the scaffold commit ships no apps/ (Prevent arrives by
// adoption) — no-op cleanly until then, as §8 step 4 requires.
if (!existsSync(APPS)) {
  console.log("verify-bindings: no apps/ yet — nothing to check");
  process.exit(0);
}

for (const app of readdirSync(APPS)) {
  const cfgPath = join(APPS, app, "wrangler.jsonc");
  if (!existsSync(cfgPath)) continue;
  // strip // comments for JSON.parse (wrangler.jsonc)
  const raw = readFileSync(cfgPath, "utf8")
    .replace(/^\s*\/\/.*$/gm, "");
  let cfg;
  try { cfg = JSON.parse(raw); }
  catch (e) { console.error(`✗ ${cfgPath}: unparseable (${e.message})`); failures++; continue; }

  const envs = ["production", ...(cfg.environments ? Object.keys(cfg.environments) : [])];
  const check = (bindings, env) => {
    for (const b of bindings?.services ?? []) {
      if (!/^[a-z-]+-worker(-[a-z]+)?$/.test(b.service ?? "")) {
        console.error(`✗ ${cfgPath} [${env}]: service binding "${b.service}" violates <app>-worker-<env> naming (NAV-006, Service bindings are non-inheritable)`);
        failures++;
      }
    }
  };
  check(cfg.bindings, "top-level");
  for (const env of envs.slice(1)) check(cfg.environments[env].bindings, env);
  console.log(`${cfgPath}: ${envs.length} environment(s) checked`);
}
if (failures) process.exit(1);
console.log("verify-bindings: clean");
```

import { frontMatterSchema } from "@my/contracts/runtime";

/** NAV-005 (Pipeline behaviour): validate + release-filter. The pipeline does NOT approve —
 *  approval is a named human in front matter; this verifies the names
 *  exist and the states are honest (NAV-005, Ownership). */
export function releaseEligible(fm: unknown): boolean {
  const parsed = frontMatterSchema.safeParse(fm);
  if (!parsed.success) return false;
  const { status, reviewer, clinical_review, release_eligible } = parsed.data;
  return status === "approved"
    && reviewer !== "pending"
    && clinical_review !== "pending"
    && release_eligible === true;
}
